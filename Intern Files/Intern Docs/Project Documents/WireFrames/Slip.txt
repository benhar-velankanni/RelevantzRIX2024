  <script>
    // Basic add/remove employee row functionality
    document.getElementById('addEmployeeBtn').addEventListener('click', () => {
      const employeeList = document.getElementById('employee-list');
      const firstRow = employeeList.querySelector('.employee-row');
      const newRow = firstRow.cloneNode(true);
      newRow.querySelectorAll('input, select').forEach(el => {
        if(el.tagName === 'SELECT') el.selectedIndex = 0;
        else el.value = el.defaultValue || '';
      });
      employeeList.appendChild(newRow);
      attachRemoveEvents();
    });

    function attachRemoveEvents(){
      document.querySelectorAll('.remove-employee-btn').forEach(btn => {
        btn.onclick = (e) => {
          const rows = document.querySelectorAll('.employee-row');
          if(rows.length > 1){
            e.target.closest('.employee-row').remove();
          } else {
            alert('At least one employee is required');
          }
        }
      });
    }

    attachRemoveEvents();
  </script>
