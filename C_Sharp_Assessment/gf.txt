CREATE TABLE Customer (
    id INT PRIMARY KEY,
    name VARCHAR(255),
    email VARCHAR(255)
);

CREATE TABLE Booking (
    id INT PRIMARY KEY,
    customer_id INT,
    service_type_id INT,
    schedule_id INT,
    FOREIGN KEY (customer_id) REFERENCES Customer(id),
    FOREIGN KEY (service_type_id) REFERENCES ServiceType(id),
    FOREIGN KEY (schedule_id) REFERENCES Schedule(id)
);

CREATE TABLE Cleaner (
    id INT PRIMARY KEY,
    name VARCHAR(255),
    email VARCHAR(255)
);

CREATE TABLE ServiceType (
    id INT PRIMARY KEY,
    name VARCHAR(255)
);

CREATE TABLE Schedule (
    id INT PRIMARY KEY,
    date DATE,
    time_slot TIME,
    cleaner_id INT,
    FOREIGN KEY (cleaner_id) REFERENCES Cleaner(id)
);
